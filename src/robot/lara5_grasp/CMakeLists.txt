cmake_minimum_required(VERSION 3.8)
project(lara5_grasp)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/grasp_node.py scripts/hand_force_check.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(configuration test/test_configuration.py)
  ament_add_pytest_test(policy test/test_policy.py)
  ament_add_pytest_test(hardware test/test_hardware.py)
  ament_add_pytest_test(force_check test/test_force_check.py)
  ament_add_pytest_test(force_check_workflow test/test_force_check_workflow.py TIMEOUT 90)
  ament_add_pytest_test(topic_workflow test/test_topic_workflow.py TIMEOUT 120)
  ament_add_pytest_test(sim_closure test/test_sim_closure.py TIMEOUT 300)
endif()
ament_package()
