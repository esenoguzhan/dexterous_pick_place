cmake_minimum_required(VERSION 3.8)
project(lara5_object_lift)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/lift_node.py DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)

  ament_add_pytest_test(sim_lift test/test_sim_lift.py TIMEOUT 300)
endif()
ament_package()
